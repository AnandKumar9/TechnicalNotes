###### Sample command to mark outlets and actions as `private`

```
class MarkOutletsAndActionsAsPrivate: NSObject, XCSourceEditorCommand {
    func perform(with invocation: XCSourceEditorCommandInvocation, completionHandler: @escaping (Error?) -> Void) {

        defer { completionHandler(nil) }

        /* All modified lines will be selected, this array will keep track of
        selections for all modified lines */
        var selectionsOfModifiedLines: [XCSourceTextRange] = []

        invocation.buffer.lines.enumerated().forEach { index, line in

            /* RegEx for all lines having outlet or action that are not marked as private already. */
            let regEx = "(IBOutlet|IBAction)(?!.*private)"
            guard
                let lineOfCode = line as? String,
                let _ = lineOfCode.range(of: regEx, options: .regularExpression)
            else { return }

            // Add private in all qualifying lines
            invocation.buffer.lines[index] = lineOfCode.replacingOccurrences(of: regEx, with: "$1 private", options: .regularExpression)

            // Add selections for all modified lines
            let lineSelection = XCSourceTextRange()
            lineSelection.start = XCSourceTextPosition(line: index, column: 0)
            lineSelection.end = XCSourceTextPosition(line: index + 1, column: 0)
            selectionsOfModifiedLines.append(lineSelection)
        }

        if selectionsOfModifiedLines.count > 0 {
            // Set the identified selections
            invocation.buffer.selections.setArray(selectionsOfModifiedLines)
        }
    }
}
```
