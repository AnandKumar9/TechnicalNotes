###### Functions to add or remove proxy settings to Terminal sessions -

Typically the system preferences' proxy settings are not in effect for terminal commands. So these functions  need to be used if required. Best to put them in bash_profile and then call them whenever needed.

```
function proxify() {
  local USERNAME
  local PASSWORD
  printf "Username: "
  read USERNAME
  stty -echo
  printf "Password: "
  read PASSWORD
  stty echo
  printf "\n"
  export http_proxy=http://${USERNAME}:${PASSWORD}@proxy.kdc.capitalone.com:8099/
  export https_proxy=$http_proxy
  export ftp_proxy=$http_proxy
  export rsync_proxy=$http_proxy
  printf "Terminal proxy has been set.\n"
}

function unproxify() {
  export http_proxy=
  export https_proxy=
  export ftp_proxy=
  export rsync_proxy=
  printf "Terminal proxy has been removed.\n"
}
```


###### Function to move n folders up -
```
go_up() { for i in $(seq $1); do cd ..; done }
go_up 3 # go 3 directories up
```
